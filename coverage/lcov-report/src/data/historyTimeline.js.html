/**
 * 根据国家 code 得到该国年表（仅包含有内容的条目）
 * @param {string} countryCode - cn | jp | uk | us | fr | de
 * @returns { { year: string, content: string }[] }
 */
export function getTimelineForCountry(countryCode) {
  const key = codeToKey[countryCode]
  if (!key) return []
  return timelineByCountry
    .filter((item) => item[key] && item[key].trim())
    .map((item) => ({ year: item.year, content: item[key] }))
}
